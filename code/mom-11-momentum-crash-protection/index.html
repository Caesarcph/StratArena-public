What the code does

Follows momentum but monitors the momentum factor's own drawdown. When momentum is experiencing a crash (reversal), reduces position size to protect against momentum crashes that historically cause severe losses.

Inputs

  • Mom_Lookback = 252
  • DD_Threshold = 0.10
  • Scale_Factor = 0.5

Indicators

  • 12-month momentum
  • Momentum factor drawdown
  • Rolling high of momentum signal

Entry

  • Momentum positive
  • Scale by momentum drawdown

Exit

  • Momentum negative or severe drawdown

def generate_signals(prices, config=None):
    config = config or {}

    mom_lookback = int(config.get("mom_lookback", 252))
    dd_threshold = float(config.get("dd_threshold", 0.10))
    severe_dd = float(config.get("severe_dd", 0.20))
    scale_factor = float(config.get("scale_factor", 0.5))
    recovery_pct = float(config.get("recovery_pct", 0.5))

    close = prices.copy()
    n = len(close)

    # Calculate momentum
    momentum = close.pct_change(mom_lookback)

    # Create a "momentum equity" - cumulative performance of momentum strategy
    # This tracks how momentum itself is doing
    mom_signal = (momentum > 0).astype(float)
    returns = close.pct_change()
    mom_returns = mom_signal.shift(1) * returns

    mom_equity = (1 + mom_returns.fillna(0)).cumprod()

    # Calculate momentum drawdown
    mom_peak = mom_equity.expanding().max()
    mom_drawdown = (mom_equity - mom_peak) / mom_peak

    signal = pd.Series(0.0, index=close.index)

    warmup = mom_lookback + 20

    for i in range(warmup, n):
        curr_mom = momentum.iloc[i]
        curr_dd = mom_drawdown.iloc[i]

        if pd.isna(curr_mom) or pd.isna(curr_dd):
            signal.iloc[i] = signal.iloc[i - 1]
            continue

        # Base signal from momentum
        if curr_mom <= 0:
            # Negative momentum - no position
            signal.iloc[i] = 0.0
            continue

        # Positive momentum - apply crash protection
        if curr_dd < -severe_dd:
            # Severe drawdown - exit completely
            signal.iloc[i] = 0.0
        elif curr_dd < -dd_threshold:
            # Moderate drawdown - scale down
            # Linear scaling from 1.0 at threshold to scale_factor at severe
            dd_range = severe_dd - dd_threshold
            dd_excess = abs(curr_dd) - dd_threshold
            scale = 1.0 - (1.0 - scale_factor) * (dd_excess / dd_range)
            signal.iloc[i] = max(scale_factor, min(1.0, scale))
        else:
            # No significant drawdown - full position
            signal.iloc[i] = 1.0

    return signal.fillna(0.0)

def _load_prices(source):
    if source.lower().endswith(".csv"):
        frame = pd.read_csv(source, index_col=0, parse_dates=True)
    else:
        import yfinance as yf
        frame = yf.download(source, start="2010-01-01", auto_adjust=True, progress=False)
        if isinstance(frame.columns, pd.MultiIndex):
            frame.columns = frame.columns.get_level_values(0)
    return frame.sort_index().dropna(subset=["Close"])


if __name__ == "__main__":
    import sys

    prices = _load_prices(sys.argv[1] if len(sys.argv) > 1 else "SPY")
    close = prices["Close"]
    signal_input = prices if STRATEGY.get("input") == "ohlc" else close
    position = generate_signals(signal_input).reindex(close.index).ffill().fillna(0.0)
    daily = close.pct_change().fillna(0.0) * position.shift(1).fillna(0.0)
    equity = (1 + daily).cumprod()
    years = (close.index[-1] - close.index[0]).days / 365.25
    print(STRATEGY["id"], STRATEGY["name"], "|", close.index[0].date(), "to", close.index[-1].date())
    print("Total return   {:+.1%}".format(equity.iloc[-1] - 1))
    print("CAGR           {:+.1%}".format(equity.iloc[-1] ** (1 / years) - 1))
    print("Sharpe         {:.2f}".format(daily.mean() / daily.std() * 252 ** 0.5 if daily.std() else 0.0))
    print("Max drawdown   {:.1%}".format((equity / equity.cummax() - 1).min()))
    print("Days in market {:.0%}".format((position != 0).mean()))
    print("Buy and hold   {:+.1%}".format(close.iloc[-1] / close.iloc[0] - 1))

momLookback = input.int(252, "Momentum lookback", minval=2)
ddThreshold = input.float(0.10, "Drawdown threshold", step=0.01)
severeDd    = input.float(0.20, "Severe drawdown", step=0.01)
scaleFactor = input.float(0.5, "Scale factor", step=0.05)
recoveryPct = input.float(0.5, "Recovery pct (unused in Python)", step=0.05)

momentum = close / close[momLookback] - 1
ret = close / close[1] - 1

// momentum-strategy equity curve: yesterday's (momentum > 0) flag times today's return
momFlagPrev = momentum[1] > 0 ? 1.0 : 0.0
var float momEquity = 1.0
momEquity := momEquity * (1 + momFlagPrev * nz(ret))
var float momPeak = 1.0
momPeak := math.max(momPeak, momEquity)
momDd = (momEquity - momPeak) / momPeak

warmup = momLookback + 20

float weight = 0.0
if bar_index >= warmup and not na(momentum)
    if momentum <= 0
        weight := 0.0
    else if momDd < -severeDd
        weight := 0.0
    else if momDd < -ddThreshold
        ddRange = severeDd - ddThreshold
        ddExcess = math.abs(momDd) - ddThreshold
        scale = 1.0 - (1.0 - scaleFactor) * (ddExcess / ddRange)
        weight := math.max(scaleFactor, math.min(1.0, scale))
    else
        weight := 1.0

plot(weight, "Target weight", color=color.teal, style=plot.style_stepline)
plot(momDd, "Momentum drawdown", color=color.red)

targetQty = close > 0 ? strategy.equity * weight / close : 0.0
gap = targetQty - strategy.position_size
if weight == 0.0
    strategy.close_all()
else if math.abs(gap) * close > 0.001 * strategy.equity
    if gap > 0
        strategy.order("Long", strategy.long, qty=gap)
    else
        strategy.order("Trim", strategy.short, qty=-gap)
