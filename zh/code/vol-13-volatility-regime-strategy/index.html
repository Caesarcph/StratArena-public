波动率状态策略：Python 与 Pine 脚本源码

这段代码做了什么

低波动时做趋势跟踪，高波动时做均值回归：低波动环境里趋势更容易延续，高波动环境里价格容易过度反应、更适合回归。

参数

  • Vol_Lookback = 20
  • Vol_Percentile_Window = 252
  • Low_Vol_Pct = 40
  • High_Vol_Pct = 70

指标

  • Realized volatility
  • Volatility percentile
  • Trend MA for low vol
  • RSI for high vol

入场

  • Low vol: trend following (price > MA)
  • High vol: mean reversion (RSI oversold)

离场

  • Regime change or signal reversal

def calculate_rsi(prices, period=14):
    delta = prices.diff()
    gain = delta.where(delta > 0, 0)
    loss = (-delta).where(delta < 0, 0)
    avg_gain = gain.rolling(period).mean()
    avg_loss = loss.rolling(period).mean()
    rs = avg_gain / (avg_loss + 1e-10)
    return 100 - (100 / (1 + rs))


def generate_signals(prices, config=None):
    config = config or {}

    vol_lookback = int(config.get("vol_lookback", 20))
    pct_window = int(config.get("pct_window", 252))
    low_vol_pct = float(config.get("low_vol_pct", 40))
    high_vol_pct = float(config.get("high_vol_pct", 70))
    trend_ma = int(config.get("trend_ma", 50))
    rsi_period = int(config.get("rsi_period", 14))
    rsi_oversold = float(config.get("rsi_oversold", 30))
    rsi_exit = float(config.get("rsi_exit", 50))

    close = prices.copy()
    n = len(close)

    # Calculate volatility
    returns = close.pct_change()
    vol = returns.rolling(vol_lookback).std() * np.sqrt(252)

    # Volatility percentile
    def vol_percentile(x):
        if len(x) < 20:
            return 50
        return (x < x.iloc[-1]).sum() / len(x) * 100

    vol_pct = vol.rolling(pct_window).apply(vol_percentile, raw=False)

    # Trend indicator for low vol regime
    ma = close.rolling(trend_ma).mean()
    uptrend = close > ma

    # Mean reversion indicator for high vol regime
    rsi = calculate_rsi(close, rsi_period)

    signal = pd.Series(0.0, index=close.index)

    warmup = max(vol_lookback, pct_window, trend_ma, rsi_period) + 10
    in_mr_trade = False

    for i in range(warmup, n):
        curr_vol_pct = vol_pct.iloc[i]
        curr_uptrend = uptrend.iloc[i]
        curr_rsi = rsi.iloc[i]

        if pd.isna(curr_vol_pct) or pd.isna(curr_rsi):
            signal.iloc[i] = signal.iloc[i - 1]
            continue

        # Determine regime
        if curr_vol_pct < low_vol_pct:
            # Low vol regime: trend following
            in_mr_trade = False
            if curr_uptrend:
                signal.iloc[i] = 1.0
            else:
                signal.iloc[i] = 0.0

        elif curr_vol_pct > high_vol_pct:
            # High vol regime: mean reversion
            if not in_mr_trade:
                # Look for entry
                if curr_rsi < rsi_oversold:
                    signal.iloc[i] = 1.0
                    in_mr_trade = True
                else:
                    signal.iloc[i] = 0.0
            else:
                # In MR trade, look for exit
                if curr_rsi > rsi_exit:
                    signal.iloc[i] = 0.0
                    in_mr_trade = False
                else:
                    signal.iloc[i] = 1.0
        else:
            # Transition zone: reduced exposure
            in_mr_trade = False
            if curr_uptrend:
                signal.iloc[i] = 0.5
            else:
                signal.iloc[i] = 0.0

    return signal.fillna(0.0)

def _load_prices(source):
    if source.lower().endswith(".csv"):
        frame = pd.read_csv(source, index_col=0, parse_dates=True)
    else:
        import yfinance as yf
        frame = yf.download(source, start="2010-01-01", auto_adjust=True, progress=False)
        if isinstance(frame.columns, pd.MultiIndex):
            frame.columns = frame.columns.get_level_values(0)
    return frame.sort_index().dropna(subset=["Close"])


if __name__ == "__main__":
    import sys

    prices = _load_prices(sys.argv[1] if len(sys.argv) > 1 else "SPY")
    close = prices["Close"]
    signal_input = prices if STRATEGY.get("input") == "ohlc" else close
    position = generate_signals(signal_input).reindex(close.index).ffill().fillna(0.0)
    daily = close.pct_change().fillna(0.0) * position.shift(1).fillna(0.0)
    equity = (1 + daily).cumprod()
    years = (close.index[-1] - close.index[0]).days / 365.25
    print(STRATEGY["id"], STRATEGY["name"], "|", close.index[0].date(), "to", close.index[-1].date())
    print("Total return   {:+.1%}".format(equity.iloc[-1] - 1))
    print("CAGR           {:+.1%}".format(equity.iloc[-1] ** (1 / years) - 1))
    print("Sharpe         {:.2f}".format(daily.mean() / daily.std() * 252 ** 0.5 if daily.std() else 0.0))
    print("Max drawdown   {:.1%}".format((equity / equity.cummax() - 1).min()))
    print("Days in market {:.0%}".format((position != 0).mean()))
    print("Buy and hold   {:+.1%}".format(close.iloc[-1] / close.iloc[0] - 1))

strategy("VOL-13 Volatility Regime Strategy", overlay=false, default_qty_type=strategy.percent_of_equity, default_qty_value=100, initial_capital=10000, commission_type=strategy.commission.percent, commission_value=0, process_orders_on_close=true)
// Position weight is 0, 0.5 or 1 (0.5 in the transition zone when in an uptrend). It is rebalanced
// to that fraction of equity every bar with strategy.order, like the Python daily-weight backtest.
// Order quantities are fractional; some instruments in TradingView may round them.
// Volatility percentile = share of the last `pct_window` vol values strictly below today's vol.

volLookback = input.int(20, "Vol lookback", minval=2)
pctWindow   = input.int(252, "Vol percentile window", minval=20)
lowVolPct   = input.float(40, "Low vol percentile")
highVolPct  = input.float(70, "High vol percentile")
trendMa     = input.int(50, "Trend MA", minval=1)
rsiPeriod   = input.int(14, "RSI period", minval=1)
rsiOversold = input.float(30, "RSI oversold")
rsiExit     = input.float(50, "RSI exit")

ret = close / close[1] - 1
vol = ta.stdev(ret, volLookback, false) * math.sqrt(252)

// full window of non-na vol values is required (pandas rolling.apply default)
float volPct = na
if not na(vol[pctWindow - 1])
    int below = 0
    for k = 0 to pctWindow - 1
        if vol[k] < vol
            below += 1
    volPct := below / pctWindow * 100

ma = ta.sma(close, trendMa)
uptrend = close > ma

delta = ta.change(close)
gain = delta > 0 ? delta : 0.0
loss = delta < 0 ? -delta : 0.0
avgGain = ta.sma(gain, rsiPeriod)
avgLoss = ta.sma(loss, rsiPeriod)
rsi = 100 - (100 / (1 + avgGain / (avgLoss + 1e-10)))

warmup = math.max(volLookback, pctWindow, trendMa, rsiPeriod) + 10

var float w = 0.0
var bool inMrTrade = false

if bar_index >= warmup and not (na(volPct) or na(rsi))
    if volPct < lowVolPct
        inMrTrade := false
        w := uptrend ? 1.0 : 0.0
    else if volPct > highVolPct
        if not inMrTrade
            if rsi < rsiOversold
                w := 1.0
                inMrTrade := true
            else
                w := 0.0
        else
            if rsi > rsiExit
                w := 0.0
                inMrTrade := false
            else
                w := 1.0
    else
        inMrTrade := false
        w := uptrend ? 0.5 : 0.0

plot(volPct, "Vol percentile", color=color.blue)
plot(w * 100, "Weight %", color=color.orange)
hline(lowVolPct, "Low vol pct", color=color.green)
hline(highVolPct, "High vol pct", color=color.red)

targetQty = close > 0 ? strategy.equity * w / close : 0.0
deltaQty = targetQty - strategy.position_size
if math.abs(deltaQty) > 1e-9
    if deltaQty > 0
        strategy.order("Rebalance", strategy.long, qty=deltaQty)
    else
        strategy.order("Rebalance", strategy.short, qty=-deltaQty)
