OU 过程均值回归：Python 与 Pine 脚本源码

OU 过程均值回归 的当前 Python 实现，附使用默认参数的示例运行器，另附 TradingView Pine 脚本版本。

下载 Python（.py）下载 Pine 脚本（.pine）查看回测结果

这段代码做了什么

对前60个收盘价拟合截距与滞后系数，不包含当前决策收盘价；只有有效且半衰期合格的均值回归拟合才允许持仓。

参数

  • lookback=60，entry_std=1.5，exit_std=.5，min_halflife=1，max_halflife=30。

指标

  • OLS斜率使用一致的协方差/方差归一化。0<b<1时，theta=−log(b)，mu=a/(1−b)，半衰期=log(2)/theta；z使用拟合的均衡波动率。

入场

  • 拟合有效、半衰期在闭区间[1,30]内，且当前z<−1.5时做多。

离场

  • z>−.5、拟合失效或半衰期离开允许区间时退出。

def estimate_ou_parameters(series):
    """Estimate OU parameters via OLS regression."""
    X = series.values
    n = len(X)

    if n < 10 or not np.isfinite(X).all():
        return None, None, None, None

    # Regression: X[t] = a + b * X[t-1]
    X_lag = X[:-1]
    X_curr = X[1:]

    # OLS estimation
    centered_lag = X_lag - X_lag.mean()
    var_lag = np.mean(centered_lag ** 2)

    if var_lag < 1e-10:
        return None, None, None, None

    # Use the same normalization for covariance and variance in the slope.
    b = np.mean(centered_lag * (X_curr - X_curr.mean())) / var_lag
    a = np.mean(X_curr) - b * np.mean(X_lag)

    residuals = X_curr - (a + b * X_lag)
    sigma_residual = np.std(residuals)

    # Convert to OU parameters (assuming dt = 1)
    if b <= 0 or b >= 1 or sigma_residual == 0:
        return None, None, None, None

    theta = -np.log(b)
    mu = a / (1 - b)
    sigma = sigma_residual * np.sqrt(-2 * np.log(b) / (1 - b**2))
    halflife = np.log(2) / theta

    return mu, theta, sigma, halflife


def generate_signals(prices, config=None):
    config = config or {}
    lookback = int(config.get("lookback", 60))
    entry_std = float(config.get("entry_std", 1.5))
    exit_std = float(config.get("exit_std", 0.5))
    min_halflife = float(config.get("min_halflife", 1))
    max_halflife = float(config.get("max_halflife", 30))

    close = prices.copy()
    n = len(close)

    signal = pd.Series(0.0, index=close.index)

    for i in range(lookback, n):
        window = close.iloc[i - lookback : i]

        # Estimate OU parameters
        mu, theta, sigma, halflife = estimate_ou_parameters(window)

        # Skip if parameters invalid or halflife out of range
        if mu is None or halflife is None:
            signal.iloc[i] = 0.0
            continue

        if halflife < min_halflife or halflife > max_halflife:
            signal.iloc[i] = 0.0
            continue

        # Equilibrium volatility
        sigma_eq = sigma / np.sqrt(2 * theta)

        # Current z-score
        current_z = (close.iloc[i] - mu) / sigma_eq

        prev_signal = signal.iloc[i - 1]

        # Trading logic
        if prev_signal == 0:
            if current_z < -entry_std:
                signal.iloc[i] = 1.0  # Buy undervalued
            else:
                signal.iloc[i] = 0.0
        else:
            if current_z > -exit_std:
                signal.iloc[i] = 0.0  # Exit when reverting
            else:
                signal.iloc[i] = 1.0

    return signal.fillna(0.0)

def _load_prices(source):
    import numpy as np

    if source.lower().endswith(".csv"):
        frame = pd.read_csv(source, index_col=0, parse_dates=True)
    else:
        import yfinance as yf
        try:
            frame = yf.download(source, start="2010-01-01", auto_adjust=True, progress=False)
        except Exception as error:
            raise ValueError("Could not download daily prices: " + str(error)) from error
        if not isinstance(frame, pd.DataFrame):
            raise ValueError("Expected a table of daily prices for one ticker.")
        if isinstance(frame.columns, pd.MultiIndex):
            levels = [level for level in range(frame.columns.nlevels)
                      if "Close" in frame.columns.get_level_values(level)]
            if len(levels) != 1:
                raise ValueError("Expected daily prices for one ticker.")
            frame.columns = frame.columns.get_level_values(levels[0])
    required = ["Open", "High", "Low", "Close"] if STRATEGY.get("input") == "ohlc" else ["Close"]
    if frame.columns.has_duplicates:
        raise ValueError("Expected one column per price field for one ticker.")
    missing = [name for name in required if name not in frame.columns]
    if missing:
        raise ValueError("Missing price columns: " + ", ".join(missing))
    if not isinstance(frame.index, pd.DatetimeIndex) or frame.index.hasnans or frame.index.normalize().has_duplicates:
        raise ValueError("Prices need one row per valid calendar date.")
    frame = frame.copy()
    frame[required] = frame[required].apply(pd.to_numeric, errors="raise")
    frame = frame.sort_index().dropna(subset=required)
    prices = frame[required].to_numpy(dtype=float)
    if len(frame) < 2:
        raise ValueError("Need at least two complete daily observations.")
    if not np.isfinite(prices).all() or (prices <= 0).any():
        raise ValueError("Prices must be finite and positive.")
    return frame


if __name__ == "__main__":
    import sys

    try:
        prices = _load_prices(sys.argv[1] if len(sys.argv) > 1 else "SPY")
    except (OSError, TypeError, ValueError) as error:
        raise SystemExit("Input error: " + str(error)) from error
    close = prices["Close"]
    signal_input = prices if STRATEGY.get("input") == "ohlc" else close
    position = generate_signals(signal_input).reindex(close.index).ffill().fillna(0.0)
    daily = close.pct_change().fillna(0.0) * position.shift(1).fillna(0.0)
    equity = (1 + daily).cumprod()
    years = (close.index[-1].date() - close.index[0].date()).days / 365.25
    print(STRATEGY["id"], STRATEGY["name"], "|", close.index[0].date(), "to", close.index[-1].date())
    print("Total return   {:+.1%}".format(equity.iloc[-1] - 1))
    print("CAGR           {:+.1%}".format(equity.iloc[-1] ** (1 / years) - 1))
    print("Sharpe         {:.2f}".format(daily.mean() / daily.std() * 252 ** 0.5 if daily.std() else 0.0))
    print("Max drawdown   {:.1%}".format((equity / equity.cummax() - 1).min()))
    print("Days in market {:.0%}".format((position != 0).mean()))
    print("Buy and hold   {:+.1%}".format(close.iloc[-1] / close.iloc[0] - 1))

// OU fit uses the previous `lookback` closes, excluding the current bar.
// The slope is the ratio of centered cross-products to centered lag squares.
strategy("MR-04 Ornstein-Uhlenbeck Mean Reversion", overlay=false, default_qty_type=strategy.percent_of_equity, default_qty_value=100, initial_capital=10000, commission_type=strategy.commission.percent, commission_value=0, process_orders_on_close=true)

lookback    = input.int(60, "Lookback", minval=12)
entryStd    = input.float(1.5, "Entry std", step=0.1)
exitStd     = input.float(0.5, "Exit std", step=0.1)
minHalflife = input.float(1, "Min half-life", step=0.5)
maxHalflife = input.float(30, "Max half-life", step=0.5)

var float pos = 0.0
float zPlot = na

if bar_index >= lookback
    // window = closes of bars i-lookback .. i-1 ; pairs (X[t], X[t-1]) -> close[k], close[k+1], k = 1 .. lookback-1
    m = lookback - 1
    sumX = 0.0
    sumY = 0.0
    for k = 1 to m
        sumY += close[k]
        sumX += close[k + 1]
    meanX = sumX / m
    meanY = sumY / m
    covSum = 0.0
    varSum = 0.0
    for k = 1 to m
        dx = close[k + 1] - meanX
        dy = close[k] - meanY
        covSum += dx * dy
        varSum += dx * dx
    varLag = varSum / m

    bool valid = false
    float mu = na
    float theta = na
    float sigma = na
    float halflife = na
    if varLag >= 1e-10
        b = covSum / varSum
        a = meanY - b * meanX
        if b > 0 and b < 1
            res = array.new_float(0)
            for k = 1 to m
                array.push(res, close[k] - (a + b * close[k + 1]))
            sigmaRes = array.stdev(res, true)
            theta := -math.log(b)
            mu := a / (1 - b)
            sigma := sigmaRes * math.sqrt(-2 * math.log(b) / (1 - b * b))
            halflife := math.log(2) / theta
            valid := sigmaRes > 0

    if not valid
        pos := 0.0
    else if halflife < minHalflife or halflife > maxHalflife
        pos := 0.0
    else
        sigmaEq = sigma / math.sqrt(2 * theta)
        currentZ = (close - mu) / sigmaEq
        zPlot := currentZ
        if pos == 0
            pos := currentZ < -entryStd ? 1.0 : 0.0
        else
            pos := currentZ > -exitStd ? 0.0 : 1.0
else
    pos := 0.0

plot(zPlot, "OU z-score", color=color.blue)
hline(-entryStd, "Entry", color=color.green)
hline(-exitStd, "Exit", color=color.gray)

if pos > 0
    strategy.entry("Long", strategy.long)
else
    strategy.close_all()
