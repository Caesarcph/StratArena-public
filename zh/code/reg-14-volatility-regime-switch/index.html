波动率状态切换：Python 与 Pine 脚本源码

波动率状态切换 的当前 Python 实现，附使用默认参数的示例运行器，另附 TradingView Pine 脚本版本。

下载 Python（.py）下载 Pine 脚本（.pine）查看回测结果

这段代码做了什么

用已实现波动率排名判断市场状态，同时以固定窗口的收盘均线确认趋势。两者共同决定每日多头仓位。

参数

  • 默认 vol_lookback = 20，percentile_window = 252，low_percentile = 50，high_percentile = 80。趋势均线固定为 50 日。

指标

  • 波动率为 vol_lookback 个日收益率的样本标准差乘 √252。排名为波动率滚动窗口中严格低于当前值的比例。
  • 收盘严格高于 50 日均线时判为上涨趋势。

入场

  • 排名低于低阈值时，上涨趋势持 100%，其余持 50%。
  • 排名介于两个阈值之间（含边界）时，上涨趋势持 75%，其余持 25%。

离场

  • 排名高于高阈值时持有现金。排名缺失时沿用此前目标。
广告

import numpy as np

STRATEGY = {'id': 'REG-14',
 'name': 'Volatility Regime Switch',
 'mode': 'single',
 'tags': ['Regime', 'Volatility', 'Adaptive'],
 'summary': 'Sizes positions from both realized-volatility rank and the 50-day price trend.',
 'description': 'This rule combines a realized-volatility percentile with a fixed '
                'closing-price SMA. The volatility regime and trend jointly determine the '
                'daily exposure.',
 'bestOn': 'Markets with volatility clustering',
 'frequency': 'Daily',
 'assetClasses': ['Index', 'Commodity', 'FX'],
 'holdingPeriod': 'Days to weeks',
 'instruments': ['SPY', 'GLD', 'USO', 'EURUSD=X'],
 'pseudocode': {'Entry': ['Below the low percentile, target 100% in an uptrend or 50% '
                          'otherwise.',
                          'Between thresholds, including equality, target 75% in an uptrend '
                          'or 25% otherwise.'],
                'Exit': ['Above the high percentile, target cash. Carry the prior target if '
                         'volatility rank is missing.'],
                'Indicators': ['Volatility is the sample standard deviation of vol_lookback '
                               'daily returns multiplied by √252. Rank is the share of the '
                               'rolling volatility window strictly below the current value.',
                               'Trend is up when the close strictly exceeds the 50-day SMA.'],
                'Inputs': ['vol_lookback = 20, percentile_window = 252, low_percentile = 50, '
                           'high_percentile = 80. The trend SMA is fixed at 50 days.'],
                'Position sizing': ['0%, 25%, 50%, 75% or 100% long.']},
 'notes': ['The rule can retain partial long exposure even when the trend test fails, '
           'depending on the volatility rank.'],
 'updated': '2026-01-11',
 'recommendedCompare': ['SPY', 'GLD']}


def generate_signals(prices, config=None):
    config = config or {}
    vol_lookback = int(config.get("vol_lookback", 20))
    percentile_window = int(config.get("percentile_window", 252))
    low_percentile = float(config.get("low_percentile", 50))
    high_percentile = float(config.get("high_percentile", 80))

    close = prices.copy()
    n = len(close)

    # Calculate realized volatility
    returns = close.pct_change()
    vol = returns.rolling(vol_lookback).std() * np.sqrt(252)

    # Calculate volatility percentile
    def percentile_rank(x):
        if len(x) < 20:
            return 50
        current = x.iloc[-1]
        return (x < current).sum() / len(x) * 100

    vol_percentile = vol.rolling(percentile_window).apply(percentile_rank, raw=False)

    # Also use trend as confirmation
    ma = close.rolling(50).mean()
    uptrend = close > ma

    signal = pd.Series(0.0, index=close.index)

    warmup = max(vol_lookback, percentile_window, 50)

    for i in range(warmup, n):
        curr_pct = vol_percentile.iloc[i]
        in_uptrend = uptrend.iloc[i]

        if pd.isna(curr_pct):
            signal.iloc[i] = signal.iloc[i - 1]
            continue

        # Regime-based position sizing
        if curr_pct < low_percentile:
            # Low vol regime: full position if uptrend
            if in_uptrend:
                signal.iloc[i] = 1.0
            else:
                signal.iloc[i] = 0.5  # Partial position even in downtrend
        elif curr_pct > high_percentile:
            # High vol regime: exit or minimal
            signal.iloc[i] = 0.0
        else:
            # Medium vol: follow trend
            if in_uptrend:
                signal.iloc[i] = 0.75
            else:
                signal.iloc[i] = 0.25

    return signal.fillna(0.0)

def _load_prices(source):
    import numpy as np

    if source.lower().endswith(".csv"):
        frame = pd.read_csv(source, index_col=0, parse_dates=True)
    else:
        import yfinance as yf
        try:
            frame = yf.download(source, start="2010-01-01", auto_adjust=True, progress=False)
        except Exception as error:
            raise ValueError("Could not download daily prices: " + str(error)) from error
        if not isinstance(frame, pd.DataFrame):
            raise ValueError("Expected a table of daily prices for one ticker.")
        if isinstance(frame.columns, pd.MultiIndex):
            levels = [level for level in range(frame.columns.nlevels)
                      if "Close" in frame.columns.get_level_values(level)]
            if len(levels) != 1:
                raise ValueError("Expected daily prices for one ticker.")
            frame.columns = frame.columns.get_level_values(levels[0])
    required = ["Open", "High", "Low", "Close"] if STRATEGY.get("input") == "ohlc" else ["Close"]
    if frame.columns.has_duplicates:
        raise ValueError("Expected one column per price field for one ticker.")
    missing = [name for name in required if name not in frame.columns]
    if missing:
        raise ValueError("Missing price columns: " + ", ".join(missing))
    if not isinstance(frame.index, pd.DatetimeIndex) or frame.index.hasnans or frame.index.normalize().has_duplicates:
        raise ValueError("Prices need one row per valid calendar date.")
    frame = frame.copy()
    frame[required] = frame[required].apply(pd.to_numeric, errors="raise")
    frame = frame.sort_index().dropna(subset=required)
    prices = frame[required].to_numpy(dtype=float)
    if len(frame) < 2:
        raise ValueError("Need at least two complete daily observations.")
    if not np.isfinite(prices).all() or (prices <= 0).any():
        raise ValueError("Prices must be finite and positive.")
    return frame


if __name__ == "__main__":
    import sys

    try:
        prices = _load_prices(sys.argv[1] if len(sys.argv) > 1 else "SPY")
    except (OSError, TypeError, ValueError) as error:
        raise SystemExit("Input error: " + str(error)) from error
    close = prices["Close"]
    signal_input = prices if STRATEGY.get("input") == "ohlc" else close
    position = generate_signals(signal_input).reindex(close.index).ffill().fillna(0.0)
    daily = close.pct_change().fillna(0.0) * position.shift(1).fillna(0.0)
    equity = (1 + daily).cumprod()
    years = (close.index[-1].date() - close.index[0].date()).days / 365.25
    print(STRATEGY["id"], STRATEGY["name"], "|", close.index[0].date(), "to", close.index[-1].date())
    print("Total return   {:+.1%}".format(equity.iloc[-1] - 1))
    print("CAGR           {:+.1%}".format(equity.iloc[-1] ** (1 / years) - 1))
    print("Sharpe         {:.2f}".format(daily.mean() / daily.std() * 252 ** 0.5 if daily.std() else 0.0))
    print("Max drawdown   {:.1%}".format((equity / equity.cummax() - 1).min()))
    print("Days in market {:.0%}".format((position != 0).mean()))
    print("Buy and hold   {:+.1%}".format(close.iloc[-1] / close.iloc[0] - 1))

// Position weight is 0, 0.25, 0.5, 0.75 or 1, rebalanced to that fraction of equity every bar with
// strategy.order, like the Python daily-weight backtest. Order quantities are fractional; some

// Volatility percentile = share of the last `percentile_window` vol values strictly below today's vol.

volLookback      = input.int(20, "Vol lookback", minval=2)
percentileWindow = input.int(252, "Percentile window", minval=20)
lowPercentile    = input.float(50, "Low percentile")
highPercentile   = input.float(80, "High percentile")

ret = close / close[1] - 1
vol = ta.stdev(ret, volLookback, false) * math.sqrt(252)

float volPct = na
if not na(vol[percentileWindow - 1])
    int below = 0
    for k = 0 to percentileWindow - 1
        if vol[k] < vol
            below += 1
    volPct := below / percentileWindow * 100

ma = ta.sma(close, 50)
uptrend = close > ma

warmup = math.max(volLookback, percentileWindow, 50)

var float w = 0.0
if bar_index >= warmup and not na(volPct)
    if volPct < lowPercentile
        w := uptrend ? 1.0 : 0.5
    else if volPct > highPercentile
        w := 0.0
    else
        w := uptrend ? 0.75 : 0.25

plot(volPct, "Vol percentile", color=color.blue)
plot(w * 100, "Weight %", color=color.orange)
hline(lowPercentile, "Low percentile", color=color.green)
hline(highPercentile, "High percentile", color=color.red)

targetQty = close > 0 ? strategy.equity * w / close : 0.0
deltaQty = targetQty - strategy.position_size
if math.abs(deltaQty) > 1e-9
    if deltaQty > 0
        strategy.order("Rebalance", strategy.long, qty=deltaQty)
    else
        strategy.order("Rebalance", strategy.short, qty=-deltaQty)
