阿隆指标：Python 与 Pine 脚本源码

阿隆指标 的当前 Python 实现，附使用默认参数的示例运行器，另附 TradingView Pine 脚本版本。

这段代码做了什么

用收盘价窗口计算 Aroon Up 和 Aroon Down。入场与退出阈值分开，建仓后可维持至指标转弱。

参数

  • 默认 period = 25，up_threshold = 70，down_threshold = 30。

指标

  • 使用当前收盘价及此前 period 个收盘价。指标为 100 ×（period − 距极值的 K 线根数）÷ period。
  • 出现相同极值时，选窗口内最早出现的位置。

入场

  • 空仓时，Aroon Up 高于 70 且 Aroon Down 低于 30 则做多。

离场

  • 持仓时，Aroon Up 低于 50 或 Aroon Down 高于 50 则平仓，其他情况持有。

import pandas as pd
import numpy as np

STRATEGY = {
    "id": "TREND-16",
    "name": "Aroon Indicator",
    "mode": "single",
    "tags": ["Trend", "Technical"],
    "summary": "Trend strength using time since highs/lows.",
    "description": "Aroon indicator measures how long since the highest high and lowest low. Aroon Up > 70 with Aroon Down < 30 indicates strong uptrend. Captures trend emergence and strength.",
    "bestOn": "Trending markets",
    "frequency": "Daily",
    "assetClasses": ["Index", "Commodity", "FX"],
    "holdingPeriod": "Days",
    "instruments": ["SPY", "GLD", "USO", "EURUSD=X"],
    "pseudocode": {
        "Inputs": ["Period = 25", "Up_Threshold = 70", "Down_Threshold = 30"],
        "Indicators": [
            "Aroon Up = ((Period - Days since highest high) / Period) * 100",
            "Aroon Down = ((Period - Days since lowest low) / Period) * 100",
        ],
        "Entry": ["Buy when Aroon Up > 70 and Aroon Down < 30"],
        "Exit": ["Sell when Aroon Up < 50 or Aroon Down > 50"],
        "Position sizing": ["Fixed 100% allocation"],
        "Risk controls": ["Exit on trend weakness"],
    },
    "notes": [
        "Developed by Tushar Chande in 1995",
        "Name means 'dawn's early light' in Sanskrit",
        "Measures trend strength and potential reversals",
    ],
    "updated": "2025-06-01",
    "recommendedCompare": ["SPY", "GLD"],
}


def generate_signals(prices, config=None):
    config = config or {}
    period = int(config.get("period", 25))
    up_threshold = float(config.get("up_threshold", 70))
    down_threshold = float(config.get("down_threshold", 30))

    close = prices.copy()

    # Calculate Aroon Up and Down
    aroon_up = pd.Series(0.0, index=close.index)
    aroon_down = pd.Series(0.0, index=close.index)

    for i in range(period, len(close)):
        window = close.iloc[i - period : i + 1]
        days_since_high = period - window.argmax()
        days_since_low = period - window.argmin()

        aroon_up.iloc[i] = ((period - days_since_high) / period) * 100
        aroon_down.iloc[i] = ((period - days_since_low) / period) * 100

    # Generate signals
    signal = pd.Series(0.0, index=close.index)
    in_position = False

    for i in range(period, len(close)):
        if not in_position:
            # Entry: strong uptrend
            if aroon_up.iloc[i] > up_threshold and aroon_down.iloc[i] < down_threshold:
                signal.iloc[i] = 1.0
                in_position = True
            else:
                signal.iloc[i] = 0.0
        else:
            # Exit: trend weakening
            if aroon_up.iloc[i] < 50 or aroon_down.iloc[i] > 50:
                signal.iloc[i] = 0.0
                in_position = False
            else:
                signal.iloc[i] = 1.0

    return signal.fillna(0.0)

def _load_prices(source):
    import numpy as np

    if source.lower().endswith(".csv"):
        frame = pd.read_csv(source, index_col=0, parse_dates=True)
    else:
        import yfinance as yf
        try:
            frame = yf.download(source, start="2010-01-01", auto_adjust=True, progress=False)
        except Exception as error:
            raise ValueError("Could not download daily prices: " + str(error)) from error
        if not isinstance(frame, pd.DataFrame):
            raise ValueError("Expected a table of daily prices for one ticker.")
        if isinstance(frame.columns, pd.MultiIndex):
            levels = [level for level in range(frame.columns.nlevels)
                      if "Close" in frame.columns.get_level_values(level)]
            if len(levels) != 1:
                raise ValueError("Expected daily prices for one ticker.")
            frame.columns = frame.columns.get_level_values(levels[0])
    required = ["Open", "High", "Low", "Close"] if STRATEGY.get("input") == "ohlc" else ["Close"]
    if frame.columns.has_duplicates:
        raise ValueError("Expected one column per price field for one ticker.")
    missing = [name for name in required if name not in frame.columns]
    if missing:
        raise ValueError("Missing price columns: " + ", ".join(missing))
    if not isinstance(frame.index, pd.DatetimeIndex) or frame.index.hasnans or frame.index.normalize().has_duplicates:
        raise ValueError("Prices need one row per valid calendar date.")
    frame = frame.copy()
    frame[required] = frame[required].apply(pd.to_numeric, errors="raise")
    frame = frame.sort_index().dropna(subset=required)
    prices = frame[required].to_numpy(dtype=float)
    if len(frame) < 2:
        raise ValueError("Need at least two complete daily observations.")
    if not np.isfinite(prices).all() or (prices <= 0).any():
        raise ValueError("Prices must be finite and positive.")
    return frame


if __name__ == "__main__":
    import sys

    try:
        prices = _load_prices(sys.argv[1] if len(sys.argv) > 1 else "SPY")
    except (OSError, TypeError, ValueError) as error:
        raise SystemExit("Input error: " + str(error)) from error
    close = prices["Close"]
    signal_input = prices if STRATEGY.get("input") == "ohlc" else close
    position = generate_signals(signal_input).reindex(close.index).ffill().fillna(0.0)
    daily = close.pct_change().fillna(0.0) * position.shift(1).fillna(0.0)
    equity = (1 + daily).cumprod()
    years = (close.index[-1].date() - close.index[0].date()).days / 365.25
    print(STRATEGY["id"], STRATEGY["name"], "|", close.index[0].date(), "to", close.index[-1].date())
    print("Total return   {:+.1%}".format(equity.iloc[-1] - 1))
    print("CAGR           {:+.1%}".format(equity.iloc[-1] ** (1 / years) - 1))
    print("Sharpe         {:.2f}".format(daily.mean() / daily.std() * 252 ** 0.5 if daily.std() else 0.0))
    print("Max drawdown   {:.1%}".format((equity / equity.cummax() - 1).min()))
    print("Days in market {:.0%}".format((position != 0).mean()))
    print("Buy and hold   {:+.1%}".format(close.iloc[-1] / close.iloc[0] - 1))

period = input.int(25, "Period", minval=1)
upThreshold = input.float(70, "Up threshold")
downThreshold = input.float(30, "Down threshold")

// Aroon on CLOSE over a window of period+1 bars; ties resolve to the OLDEST extreme (numpy argmax/argmin)
float aroonUp = 0.0
float aroonDown = 0.0
if bar_index >= period
    float hi = close
    float lo = close
    int hk = 0
    int lk = 0
    for k = 1 to period
        if close[k] >= hi
            hi := close[k]
            hk := k
        if close[k] <= lo
            lo := close[k]
            lk := k
    aroonUp := (period - hk) / period * 100.0
    aroonDown := (period - lk) / period * 100.0

var bool inPos = false
if bar_index >= period
    if not inPos
        if aroonUp > upThreshold and aroonDown < downThreshold
            inPos := true
    else
        if aroonUp < 50 or aroonDown > 50
            inPos := false

plot(aroonUp, "Aroon Up", color=color.green)
plot(aroonDown, "Aroon Down", color=color.red)
hline(70, "70", color=color.gray)
hline(30, "30", color=color.gray)
hline(50, "50", color=color.silver)

if inPos
    strategy.entry("Long", strategy.long)
else
    strategy.close_all()
