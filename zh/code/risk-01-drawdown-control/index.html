回撤控制：Python 与 Pine 脚本源码

本站 回撤控制 回测背后的那份 Python 文件，另附 TradingView Pine 脚本版本。

这段代码做了什么

实时监控组合回撤，回撤超过阈值时按比例减仓，类似 CPPI 动态保险，可以叠加在任何策略上。

参数

  • Max_Drawdown = 0.10
  • Scale = 0.5
  • Recovery_Threshold = 0.5

指标

  • Rolling maximum price
  • Current drawdown

入场

  • Full position when drawdown < threshold
  • Reduce by scale factor when drawdown exceeds

离场

  • Gradually restore as drawdown recovers
广告

def generate_signals(prices, config=None):
    config = config or {}
    max_drawdown = float(config.get("max_drawdown", 0.10))
    scale_factor = float(config.get("scale", 0.5))
    lookback = int(config.get("lookback", 252))
    min_position = float(config.get("min_position", 0.1))

    close = prices.copy()
    n = len(close)

    # Calculate rolling maximum and drawdown
    rolling_max = close.rolling(lookback, min_periods=1).max()
    drawdown = (close - rolling_max) / rolling_max

    signal = pd.Series(1.0, index=close.index)  # Start fully invested

    for i in range(1, n):
        current_dd = abs(drawdown.iloc[i])

        if current_dd <= max_drawdown:
            # No significant drawdown - full position
            signal.iloc[i] = 1.0
        else:
            # Drawdown exceeds threshold - reduce position
            excess_dd = current_dd - max_drawdown

            # Linear reduction based on excess drawdown
            reduction = min(excess_dd / max_drawdown * scale_factor, 1 - min_position)
            signal.iloc[i] = max(1.0 - reduction, min_position)

    return signal.fillna(1.0)

def _load_prices(source):
    if source.lower().endswith(".csv"):
        frame = pd.read_csv(source, index_col=0, parse_dates=True)
    else:
        import yfinance as yf
        frame = yf.download(source, start="2010-01-01", auto_adjust=True, progress=False)
        if isinstance(frame.columns, pd.MultiIndex):
            frame.columns = frame.columns.get_level_values(0)
    return frame.sort_index().dropna(subset=["Close"])


if __name__ == "__main__":
    import sys

    prices = _load_prices(sys.argv[1] if len(sys.argv) > 1 else "SPY")
    close = prices["Close"]
    signal_input = prices if STRATEGY.get("input") == "ohlc" else close
    position = generate_signals(signal_input).reindex(close.index).ffill().fillna(0.0)
    daily = close.pct_change().fillna(0.0) * position.shift(1).fillna(0.0)
    equity = (1 + daily).cumprod()
    years = (close.index[-1] - close.index[0]).days / 365.25
    print(STRATEGY["id"], STRATEGY["name"], "|", close.index[0].date(), "to", close.index[-1].date())
    print("Total return   {:+.1%}".format(equity.iloc[-1] - 1))
    print("CAGR           {:+.1%}".format(equity.iloc[-1] ** (1 / years) - 1))
    print("Sharpe         {:.2f}".format(daily.mean() / daily.std() * 252 ** 0.5 if daily.std() else 0.0))
    print("Max drawdown   {:.1%}".format((equity / equity.cummax() - 1).min()))
    print("Days in market {:.0%}".format((position != 0).mean()))
    print("Buy and hold   {:+.1%}".format(close.iloc[-1] / close.iloc[0] - 1))

// Fractional position: the Python returns a weight between min_position and 1.0 that shrinks as the drawdown from the rolling
// high grows. This script resizes the long position with strategy.order to that fraction of equity whenever the target weight
// changes (no rebalance while the weight is unchanged).
strategy("RISK-01 Drawdown Control", overlay=false, default_qty_type=strategy.percent_of_equity, default_qty_value=100, initial_capital=10000, commission_type=strategy.commission.percent, commission_value=0, process_orders_on_close=true)

maxDrawdown = input.float(0.10, "Max drawdown", minval=0.0001)
scaleFactor = input.float(0.5, "Scale")
lookback = input.int(252, "Lookback", minval=1)
minPosition = input.float(0.1, "Min position")

// rolling max with min_periods=1: window shrinks to the available history
rollingMax = ta.highest(close, math.min(lookback, bar_index + 1))
currentDd = math.abs((close - rollingMax) / rollingMax)

float w = 1.0
if currentDd > maxDrawdown
    excessDd = currentDd - maxDrawdown
    reduction = math.min(excessDd / maxDrawdown * scaleFactor, 1 - minPosition)
    w := math.max(1.0 - reduction, minPosition)

plot(w, "Target weight", color=color.orange)
plot(currentDd, "Drawdown", color=color.red)

var float lastW = 0.0
if w != lastW
    if w <= 0
        strategy.close_all()
    else
        targetQty = strategy.equity * w / close
        delta = targetQty - strategy.position_size
        if math.abs(delta) > 0
            strategy.order("Long", delta > 0 ? strategy.long : strategy.short, qty=math.abs(delta))
    lastW := w
