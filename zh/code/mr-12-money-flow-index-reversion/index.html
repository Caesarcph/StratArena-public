资金流量指标回归：Python 与 Pine 脚本源码

本站 资金流量指标回归 回测背后的那份 Python 文件，另附 TradingView Pine 脚本版本。

这段代码做了什么

用资金流量指标（MFI，相当于成交量加权的 RSI）识别超卖。MFI < 20 表示有成交量确认的超卖，MFI 回到中性时离场。

参数

  • Period = 14
  • Oversold = 30
  • Exit = 50

指标

  • Typical Price = (High + Low + Close) / 3
  • Raw Money Flow = Typical Price * Volume
  • MFI = 100 - 100 / (1 + Positive MF / Negative MF)

入场

  • Buy when MFI < 30 (oversold)

离场

  • Exit when MFI > 50 (neutral)
广告

def generate_signals(prices, config=None):
    """
    Note: Without volume data, we approximate MFI using price momentum.
    In production, this would use actual volume data.
    """
    config = config or {}
    period = int(config.get("period", 14))
    oversold = float(config.get("oversold", 30))  # Relaxed from 20 for more signals
    exit_level = float(config.get("exit_level", 50))
    max_hold = int(config.get("max_hold", 15))

    close = prices.copy()

    # Approximate MFI using RSI-like calculation
    # (In practice, would use volume data)
    delta = close.diff()
    gain = delta.where(delta > 0, 0)
    loss = (-delta).where(delta < 0, 0)

    avg_gain = gain.rolling(period).mean()
    avg_loss = loss.rolling(period).mean()

    rs = avg_gain / (avg_loss + 1e-10)
    mfi_approx = 100 - (100 / (1 + rs))

    signal = pd.Series(0.0, index=close.index)
    entry_idx = None

    for i in range(period, len(close)):
        prev_signal = signal.iloc[i - 1]

        if prev_signal == 0:
            # Check for oversold entry
            if mfi_approx.iloc[i] < oversold:
                signal.iloc[i] = 1.0
                entry_idx = i
            else:
                signal.iloc[i] = 0.0
        else:
            # In position - check for exit
            hold_days = i - entry_idx if entry_idx else 0

            # Exit on recovery or max hold
            if mfi_approx.iloc[i] > exit_level:
                signal.iloc[i] = 0.0
            elif hold_days >= max_hold:
                signal.iloc[i] = 0.0
            else:
                signal.iloc[i] = 1.0

    return signal.fillna(0.0)

def _load_prices(source):
    if source.lower().endswith(".csv"):
        frame = pd.read_csv(source, index_col=0, parse_dates=True)
    else:
        import yfinance as yf
        frame = yf.download(source, start="2010-01-01", auto_adjust=True, progress=False)
        if isinstance(frame.columns, pd.MultiIndex):
            frame.columns = frame.columns.get_level_values(0)
    return frame.sort_index().dropna(subset=["Close"])


if __name__ == "__main__":
    import sys

    prices = _load_prices(sys.argv[1] if len(sys.argv) > 1 else "SPY")
    close = prices["Close"]
    signal_input = prices if STRATEGY.get("input") == "ohlc" else close
    position = generate_signals(signal_input).reindex(close.index).ffill().fillna(0.0)
    daily = close.pct_change().fillna(0.0) * position.shift(1).fillna(0.0)
    equity = (1 + daily).cumprod()
    years = (close.index[-1] - close.index[0]).days / 365.25
    print(STRATEGY["id"], STRATEGY["name"], "|", close.index[0].date(), "to", close.index[-1].date())
    print("Total return   {:+.1%}".format(equity.iloc[-1] - 1))
    print("CAGR           {:+.1%}".format(equity.iloc[-1] ** (1 / years) - 1))
    print("Sharpe         {:.2f}".format(daily.mean() / daily.std() * 252 ** 0.5 if daily.std() else 0.0))
    print("Max drawdown   {:.1%}".format((equity / equity.cummax() - 1).min()))
    print("Days in market {:.0%}".format((position != 0).mean()))
    print("Buy and hold   {:+.1%}".format(close.iloc[-1] / close.iloc[0] - 1))

strategy("MR-12 Money Flow Index Reversion", overlay=false, default_qty_type=strategy.percent_of_equity, default_qty_value=100, initial_capital=10000, commission_type=strategy.commission.percent, commission_value=0, process_orders_on_close=true)

period    = input.int(14, "Period", minval=1)
oversold  = input.float(30, "Oversold")
exitLevel = input.float(50, "Exit level")
maxHold   = input.int(15, "Max hold (bars)", minval=1)

delta = ta.change(close)
gain = delta > 0 ? delta : 0.0
loss = delta < 0 ? -delta : 0.0
avgGain = ta.sma(gain, period)
avgLoss = ta.sma(loss, period)
rs = avgGain / (avgLoss + 1e-10)
mfiApprox = 100 - (100 / (1 + rs))

var float pos = 0.0
var int entryBar = na

if bar_index >= period
    if pos == 0
        if mfiApprox < oversold
            pos := 1.0
            entryBar := bar_index
        else
            pos := 0.0
    else
        holdDays = not na(entryBar) ? bar_index - entryBar : 0
        if mfiApprox > exitLevel
            pos := 0.0
        else if holdDays >= maxHold
            pos := 0.0
        else
            pos := 1.0
else
    pos := 0.0

plot(mfiApprox, "MFI approx (RSI-like)", color=color.purple)
hline(oversold, "Oversold", color=color.green)
hline(exitLevel, "Exit", color=color.gray)

if pos > 0
    strategy.entry("Long", strategy.long)
else
    strategy.close_all()
