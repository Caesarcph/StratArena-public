月末月初效应：Python 与 Pine 脚本源码

本站 月末月初效应 回测背后的那份 Python 文件，另附 TradingView Pine 脚本版本。

这段代码做了什么

采用周一至周五的日历计数：每月已过工作日不超过 Start_Days，或剩余工作日少于 End_Days 时持有。它不使用未来下载的数据，也不等同于交易所假日日历。

参数

  • End_Days = 3
  • Start_Days = 3

指标

  • Elapsed/remaining Monday-Friday calendar weekdays

入场

  • Hold when at most Start_Days calendar weekdays have elapsed
  • Hold when fewer than End_Days calendar weekdays remain

离场

  • Exit outside turn-of-month window
广告

def get_trading_days_info(dates):
    """Known weekday counts, independent of future downloaded observations.

    The function name is retained for compatibility. These are calendar
    approximations, not actual exchange-session or retrospective closure counts.
    """
    days = np.asarray(pd.DatetimeIndex(dates).date, dtype="datetime64[D]")
    months = days.astype("datetime64[M]")
    first = months.astype("datetime64[D]")
    following = (months + 1).astype("datetime64[D]")
    next_day = days + np.timedelta64(1, "D")
    return np.busday_count(first, next_day), np.busday_count(next_day, following)


def generate_signals(prices, config=None):
    config = config or {}
    end_days = int(config.get("end_days", 3))
    start_days = int(config.get("start_days", 3))

    close = prices.copy()

    # Get trading day info
    day_of_month, days_until_end = get_trading_days_info(close.index)

    signal = pd.Series(0.0, index=close.index)

    for i in range(len(close)):
        # First N days of month
        is_month_start = start_days > 0 and day_of_month[i] <= start_days

        # Last M days of month
        is_month_end = end_days > 0 and days_until_end[i] < end_days

        if is_month_start or is_month_end:
            signal.iloc[i] = 1.0
        else:
            signal.iloc[i] = 0.0

    return signal.fillna(0.0)

# Convention, same as the site: the position decided at the close of day t earns

def _load_prices(source):
    if source.lower().endswith(".csv"):
        frame = pd.read_csv(source, index_col=0, parse_dates=True)
    else:
        import yfinance as yf
        frame = yf.download(source, start="2010-01-01", auto_adjust=True, progress=False)
        if isinstance(frame.columns, pd.MultiIndex):
            frame.columns = frame.columns.get_level_values(0)
    return frame.sort_index().dropna(subset=["Close"])


if __name__ == "__main__":
    import sys

    prices = _load_prices(sys.argv[1] if len(sys.argv) > 1 else "SPY")
    close = prices["Close"]
    signal_input = prices if STRATEGY.get("input") == "ohlc" else close
    position = generate_signals(signal_input).reindex(close.index).ffill().fillna(0.0)
    daily = close.pct_change().fillna(0.0) * position.shift(1).fillna(0.0)
    equity = (1 + daily).cumprod()
    years = (close.index[-1] - close.index[0]).days / 365.25
    print(STRATEGY["id"], STRATEGY["name"], "|", close.index[0].date(), "to", close.index[-1].date())
    print("Total return   {:+.1%}".format(equity.iloc[-1] - 1))
    print("CAGR           {:+.1%}".format(equity.iloc[-1] ** (1 / years) - 1))
    print("Sharpe         {:.2f}".format(daily.mean() / daily.std() * 252 ** 0.5 if daily.std() else 0.0))
    print("Max drawdown   {:.1%}".format((equity / equity.cummax() - 1).min()))
    print("Days in market {:.0%}".format((position != 0).mean()))
    print("Buy and hold   {:+.1%}".format(close.iloc[-1] / close.iloc[0] - 1))

strategy("CAL-01 Turn of Month", overlay=false, default_qty_type=strategy.percent_of_equity, default_qty_value=100, initial_capital=10000, commission_type=strategy.commission.percent, commission_value=0, process_orders_on_close=true)
// Fixed Monday-Friday calendar (no exchange holidays), as in the Python implementation:
//   elapsed   = weekdays from the 1st of the month up to and including today
//   remaining = weekdays after today until the end of the month
// Long when elapsed <= Start days, or remaining < End days.

endDays   = input.int(3, "End days", minval=0)
startDays = input.int(3, "Start days", minval=0)

isLeap = (year % 4 == 0 and year % 100 != 0) or (year % 400 == 0)
dim = switch month
    2 => isLeap ? 29 : 28
    4 => 30
    6 => 30
    9 => 30
    11 => 30
    => 31

dom = dayofmonth
todayIdx = dayofweek - 1   // 0 = Sunday ... 6 = Saturday

int elapsed = 0
int remaining = 0
for k = 1 to dim
    dIdx = ((todayIdx - (dom - k)) % 7 + 7) % 7
    isWeekday = dIdx >= 1 and dIdx <= 5
    if isWeekday
        if k <= dom
            elapsed += 1
        else
            remaining += 1

isMonthStart = startDays > 0 and elapsed <= startDays
isMonthEnd = endDays > 0 and remaining < endDays
pos = (isMonthStart or isMonthEnd) ? 1 : 0

plot(elapsed, "Weekdays elapsed", color=color.blue)
plot(remaining, "Weekdays remaining", color=color.orange)
bgcolor(pos > 0 ? color.new(color.green, 88) : na)

if pos > 0
    strategy.entry("Long", strategy.long)
else
    strategy.close_all()
